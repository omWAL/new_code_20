# Simple Customer Chatbot in C++

## Simple C++ Code

```cpp id="llv9jg"
#include <iostream>
#include <string>
using namespace std;

int main()
{
    string message;

    cout << "Customer Chatbot\n";

    while (true)
    {
        cout << "\nYou: ";
        getline(cin, message);

        if (message == "hello")
        {
            cout << "Bot: Hello! How can I help you?";
        }

        else if (message == "price")
        {
            cout << "Bot: Product price is 500 rupees.";
        }

        else if (message == "order")
        {
            cout << "Bot: Your order will arrive in 3 days.";
        }

        else if (message == "bye")
        {
            cout << "Bot: Thank you!";
            break;
        }

        else
        {
            cout << "Bot: Sorry, I don't understand.";
        }
    }

    return 0;
}
```
